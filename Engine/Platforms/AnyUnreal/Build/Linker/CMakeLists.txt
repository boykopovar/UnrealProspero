cmake_minimum_required(VERSION 3.22)
project(AnyUnrealLink LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
include("${ANYUNREAL_MANIFEST}")
foreach(object IN LISTS ANYUNREAL_OBJECTS)
    if(NOT EXISTS "${object}")
        message(FATAL_ERROR "Missing link input: ${object}")
    endif()
    set_source_files_properties("${object}" PROPERTIES EXTERNAL_OBJECT TRUE)
endforeach()
adk_add_executable(AnyUnrealApplication ${ANYUNREAL_OBJECTS})
set_target_properties(AnyUnrealApplication PROPERTIES LINKER_LANGUAGE CXX OUTPUT_NAME "${ANYUNREAL_NAME}" RUNTIME_OUTPUT_DIRECTORY "${ANYUNREAL_OUTPUT_DIRECTORY}")
adk_link_libraries(AnyUnrealApplication ${ANYUNREAL_IMPORTS})
target_link_libraries(AnyUnrealApplication PRIVATE ${ANYUNREAL_LIBRARIES})
target_link_options(AnyUnrealApplication PRIVATE ${ANYUNREAL_LINK_OPTIONS})
adk_runtime_libraries(AnyUnrealApplication ${ANYUNREAL_RUNTIME_LIBRARIES})
