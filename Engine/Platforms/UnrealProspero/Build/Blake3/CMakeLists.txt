cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoBlake3 LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT DEFINED BLAKE3_SOURCE_DIRECTORY OR NOT EXISTS "${BLAKE3_SOURCE_DIRECTORY}/blake3.h")
    message(FATAL_ERROR "Set BLAKE3_SOURCE_DIRECTORY to the BLAKE3 C sources supplied with Unreal")
endif()
set(sources blake3 blake3_dispatch blake3_portable blake3_sse2 blake3_sse41 blake3_avx2 blake3_avx512)
list(TRANSFORM sources PREPEND "${BLAKE3_SOURCE_DIRECTORY}/")
list(TRANSFORM sources APPEND ".c")
add_library(UnrealProsperoBlake3 STATIC ${sources})
set_target_properties(UnrealProsperoBlake3 PROPERTIES OUTPUT_NAME BLAKE3 POSITION_INDEPENDENT_CODE ON C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(UnrealProsperoBlake3 PUBLIC "${BLAKE3_SOURCE_DIRECTORY}")
target_link_libraries(UnrealProsperoBlake3 PRIVATE AnyLibcxx::Runtime)
target_compile_options(UnrealProsperoBlake3 PRIVATE -fno-builtin -fno-stack-protector)
set_source_files_properties("${BLAKE3_SOURCE_DIRECTORY}/blake3_sse2.c" PROPERTIES COMPILE_OPTIONS -msse2)
set_source_files_properties("${BLAKE3_SOURCE_DIRECTORY}/blake3_sse41.c" PROPERTIES COMPILE_OPTIONS -msse4.1)
set_source_files_properties("${BLAKE3_SOURCE_DIRECTORY}/blake3_avx2.c" PROPERTIES COMPILE_OPTIONS -mavx2)
set_source_files_properties("${BLAKE3_SOURCE_DIRECTORY}/blake3_avx512.c" PROPERTIES COMPILE_OPTIONS "-mavx512f;-mavx512vl")
install(TARGETS UnrealProsperoBlake3 ARCHIVE DESTINATION lib)
