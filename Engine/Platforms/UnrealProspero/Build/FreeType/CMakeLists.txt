cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoFreeType LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
foreach(input FREETYPE_SOURCE_DIRECTORY PNG_PNG_INCLUDE_DIR PNG_LIBRARY ZLIB_INCLUDE_DIR ZLIB_LIBRARY)
    if(NOT DEFINED ${input} OR NOT EXISTS "${${input}}")
        message(FATAL_ERROR "Set ${input} to the matching FreeType source or AnyDevKit dependency")
    endif()
endforeach()
foreach(template configure.raw ftconfig.h.in freetype2.in)
    if(NOT EXISTS "${FREETYPE_SOURCE_DIRECTORY}/builds/unix/${template}")
        message(FATAL_ERROR "FreeType source is missing builds/unix/${template}; provide the complete FreeType 2.14.1 source")
    endif()
endforeach()
set(BUILD_SHARED_LIBS OFF)
set(FT_REQUIRE_ZLIB ON)
set(FT_REQUIRE_PNG ON)
set(FT_DISABLE_BZIP2 ON)
set(FT_DISABLE_HARFBUZZ ON)
set(FT_DISABLE_BROTLI ON)
add_subdirectory("${FREETYPE_SOURCE_DIRECTORY}" freetype)
if(ADK_TARGET STREQUAL "ps5")
    get_target_property(freetypeSources freetype SOURCES)
    list(FIND freetypeSources "builds/unix/ftsystem.c" unixSystem)
    if(unixSystem EQUAL -1)
        message(FATAL_ERROR "FreeType did not select its expected Unix file implementation")
    endif()
    list(REMOVE_AT freetypeSources ${unixSystem})
    list(APPEND freetypeSources "${FREETYPE_SOURCE_DIRECTORY}/src/base/ftsystem.c")
    set_property(TARGET freetype PROPERTY SOURCES "${freetypeSources}")
endif()
set_target_properties(freetype PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_link_libraries(freetype PRIVATE AnyLibcxx::Runtime)
target_compile_options(freetype PRIVATE -fno-builtin -fno-stack-protector)
