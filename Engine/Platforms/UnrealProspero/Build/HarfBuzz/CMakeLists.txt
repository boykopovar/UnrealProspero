cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoHarfBuzz LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT EXISTS "${HARFBUZZ_SOURCE_DIRECTORY}/hb-ft.cc" OR NOT EXISTS "${FREETYPE_INCLUDE_DIRECTORY}/ft2build.h" OR NOT EXISTS "${ICU_INCLUDE_DIRECTORY}/unicode/utypes.h")
    message(FATAL_ERROR "Set HARFBUZZ_SOURCE_DIRECTORY, FREETYPE_INCLUDE_DIRECTORY and ICU_INCLUDE_DIRECTORY to the Unreal HarfBuzz 2.4, FreeType and ICU 64 inputs")
endif()
file(GLOB sources CONFIGURE_DEPENDS "${HARFBUZZ_SOURCE_DIRECTORY}/*.cc" "${HARFBUZZ_SOURCE_DIRECTORY}/*.c")
list(FILTER sources EXCLUDE REGEX "/(main|dump-[^/]*|test[^/]*|hb-(coretext|directwrite|glib|gobject-structs|graphite2|ucdn|uniscribe))\\.cc$")
add_library(UnrealProsperoHarfBuzz STATIC ${sources})
set_target_properties(UnrealProsperoHarfBuzz PROPERTIES OUTPUT_NAME harfbuzz POSITION_INDEPENDENT_CODE ON CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(UnrealProsperoHarfBuzz SYSTEM PRIVATE "${FREETYPE_INCLUDE_DIRECTORY}" "${ICU_INCLUDE_DIRECTORY}")
target_link_libraries(UnrealProsperoHarfBuzz PRIVATE AnyLibcxx::Runtime)
target_compile_definitions(UnrealProsperoHarfBuzz PRIVATE HAVE_ROUND HAVE_OT HAVE_FALLBACK HAVE_ICU HAVE_ICU_BUILTIN U_STATIC_IMPLEMENTATION)
target_compile_options(UnrealProsperoHarfBuzz PRIVATE -fno-builtin -fno-stack-protector)
install(TARGETS UnrealProsperoHarfBuzz ARCHIVE DESTINATION lib)
