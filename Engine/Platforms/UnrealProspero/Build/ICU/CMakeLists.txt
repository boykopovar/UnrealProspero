cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoICU LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT DEFINED ICU_SOURCE_DIRECTORY OR NOT EXISTS "${ICU_SOURCE_DIRECTORY}/common/unicode/utypes.h" OR NOT EXISTS "${ICU_SOURCE_DIRECTORY}/i18n/unicode/ucol.h" OR NOT EXISTS "${ICU_SOURCE_DIRECTORY}/stubdata/stubdata.cpp")
    message(FATAL_ERROR "Set ICU_SOURCE_DIRECTORY to the ICU 64 source directory supplied with Unreal")
endif()
file(GLOB common CONFIGURE_DEPENDS "${ICU_SOURCE_DIRECTORY}/common/*.cpp" "${ICU_SOURCE_DIRECTORY}/common/*.c")
file(GLOB i18n CONFIGURE_DEPENDS "${ICU_SOURCE_DIRECTORY}/i18n/*.cpp" "${ICU_SOURCE_DIRECTORY}/i18n/*.c")
add_library(UnrealProsperoICU STATIC ${common} ${i18n} "${ICU_SOURCE_DIRECTORY}/stubdata/stubdata.cpp")
set_target_properties(UnrealProsperoICU PROPERTIES OUTPUT_NAME icu POSITION_INDEPENDENT_CODE ON CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(UnrealProsperoICU PRIVATE "${ICU_SOURCE_DIRECTORY}/common" "${ICU_SOURCE_DIRECTORY}/i18n")
target_link_libraries(UnrealProsperoICU PRIVATE AnyLibcxx::Runtime)
target_compile_definitions(UnrealProsperoICU PRIVATE U_STATIC_IMPLEMENTATION U_COMMON_IMPLEMENTATION U_I18N_IMPLEMENTATION U_USING_ICU_NAMESPACE=0 UCONFIG_NO_TRANSLITERATION=1)
if(ADK_TARGET STREQUAL "ps5")
    target_compile_definitions(UnrealProsperoICU PRIVATE LC_MESSAGES=32)
    set_source_files_properties("${ICU_SOURCE_DIRECTORY}/common/putil.cpp" PROPERTIES COMPILE_DEFINITIONS "U_HAVE_TIMEZONE=0;U_HAVE_TZNAME=0;DEBUG_SKIP_LOCALTIME_LINK")
    set_source_files_properties("${ICU_SOURCE_DIRECTORY}/common/umapfile.cpp" PROPERTIES COMPILE_DEFINITIONS U_HAVE_MMAP=0)
endif()
target_compile_options(UnrealProsperoICU PRIVATE -fno-builtin -fno-stack-protector)
install(TARGETS UnrealProsperoICU ARCHIVE DESTINATION lib)
