cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoLibPng LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT DEFINED PNG_SOURCE_DIRECTORY OR NOT EXISTS "${PNG_SOURCE_DIRECTORY}/pnglibconf.h")
    message(FATAL_ERROR "Set PNG_SOURCE_DIRECTORY to the libPNG sources supplied with Unreal")
endif()
if(NOT DEFINED ZLIB_SOURCE_DIRECTORY OR NOT EXISTS "${ZLIB_SOURCE_DIRECTORY}/zlib.h")
    message(FATAL_ERROR "Set ZLIB_SOURCE_DIRECTORY to the zlib sources supplied with Unreal")
endif()
set(sources png pngerror pngget pngmem pngpread pngread pngrio pngrtran pngrutil pngset pngtrans pngwio pngwrite pngwtran pngwutil)
list(TRANSFORM sources PREPEND "${PNG_SOURCE_DIRECTORY}/")
list(TRANSFORM sources APPEND ".c")
add_library(UnrealProsperoLibPng STATIC ${sources})
set_target_properties(UnrealProsperoLibPng PROPERTIES OUTPUT_NAME png POSITION_INDEPENDENT_CODE ON C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(UnrealProsperoLibPng PUBLIC "${PNG_SOURCE_DIRECTORY}" PRIVATE "${ZLIB_SOURCE_DIRECTORY}")
target_link_libraries(UnrealProsperoLibPng PRIVATE AnyLibcxx::Runtime)
target_compile_definitions(UnrealProsperoLibPng PRIVATE PNG_INTEL_SSE_OPT=0)
target_compile_options(UnrealProsperoLibPng PRIVATE -fno-builtin -fno-stack-protector)
install(TARGETS UnrealProsperoLibPng ARCHIVE DESTINATION lib)
