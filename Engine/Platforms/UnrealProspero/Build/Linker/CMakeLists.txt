cmake_minimum_required(VERSION 3.22)
project(UnrealProsperoLink LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
include("${UNREALPROSPERO_MANIFEST}")
foreach(object IN LISTS UNREALPROSPERO_OBJECTS)
    if(NOT EXISTS "${object}")
        message(FATAL_ERROR "Missing link input: ${object}")
    endif()
    set_source_files_properties("${object}" PROPERTIES EXTERNAL_OBJECT TRUE)
endforeach()
adk_add_executable(UnrealProsperoApplication ${UNREALPROSPERO_OBJECTS})
set_target_properties(UnrealProsperoApplication PROPERTIES LINKER_LANGUAGE CXX OUTPUT_NAME "${UNREALPROSPERO_NAME}" RUNTIME_OUTPUT_DIRECTORY "${UNREALPROSPERO_OUTPUT_DIRECTORY}")
get_target_property(sdkLibraries UnrealProsperoApplication LINK_LIBRARIES)
set(linkLibraries ${UNREALPROSPERO_LIBRARIES})
list(APPEND linkLibraries ${sdkLibraries})
set_property(TARGET UnrealProsperoApplication PROPERTY LINK_LIBRARIES "${linkLibraries}")
adk_link_libraries(UnrealProsperoApplication ${UNREALPROSPERO_IMPORTS})
if(ADK_TARGET STREQUAL "ps5")
    target_link_options(UnrealProsperoApplication PRIVATE -Wl,--no-export-dynamic,--export-dynamic-symbol=main)
endif()
target_link_options(UnrealProsperoApplication PRIVATE ${UNREALPROSPERO_LINK_OPTIONS})
adk_runtime_libraries(UnrealProsperoApplication ${UNREALPROSPERO_RUNTIME_LIBRARIES})
