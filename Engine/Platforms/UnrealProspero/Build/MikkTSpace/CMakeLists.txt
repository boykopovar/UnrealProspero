cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoMikkTSpace LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT EXISTS "${MIKKTSPACE_SOURCE_DIRECTORY}/src/mikktspace.c" OR NOT EXISTS "${MIKKTSPACE_SOURCE_DIRECTORY}/inc/mikktspace.h")
    message(FATAL_ERROR "Set MIKKTSPACE_SOURCE_DIRECTORY to the MikkTSpace sources supplied with Unreal")
endif()
add_library(UnrealProsperoMikkTSpace STATIC "${MIKKTSPACE_SOURCE_DIRECTORY}/src/mikktspace.c")
set_target_properties(UnrealProsperoMikkTSpace PROPERTIES OUTPUT_NAME MikkTSpace POSITION_INDEPENDENT_CODE ON C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
target_include_directories(UnrealProsperoMikkTSpace PUBLIC "${MIKKTSPACE_SOURCE_DIRECTORY}/inc")
target_link_libraries(UnrealProsperoMikkTSpace PRIVATE AnyLibcxx::Runtime)
target_compile_options(UnrealProsperoMikkTSpace PRIVATE -fno-builtin -fno-stack-protector)
install(TARGETS UnrealProsperoMikkTSpace ARCHIVE DESTINATION lib)
