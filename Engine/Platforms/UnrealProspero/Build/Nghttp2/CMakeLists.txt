cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoNghttp2 LANGUAGES C CXX)

find_package(AnyDevKit CONFIG REQUIRED)
if(NOT ADK_TARGET STREQUAL "ps5")
    message(FATAL_ERROR "The nghttp2 archive requires the AnyDevKit PS5 toolchain")
endif()
if(NOT DEFINED NGHTTP2_SOURCE_DIRECTORY OR NOT EXISTS "${NGHTTP2_SOURCE_DIRECTORY}/lib/nghttp2_session.c")
    message(FATAL_ERROR "Set NGHTTP2_SOURCE_DIRECTORY to the nghttp2 source directory")
endif()

set(ENABLE_LIB_ONLY ON)
set(BUILD_SHARED_LIBS OFF)
set(BUILD_STATIC_LIBS ON)
set(BUILD_TESTING OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory("${NGHTTP2_SOURCE_DIRECTORY}" "${CMAKE_CURRENT_BINARY_DIR}/nghttp2" EXCLUDE_FROM_ALL)
target_compile_options(nghttp2_static PRIVATE -fno-builtin -fno-stack-protector)
install(TARGETS nghttp2_static ARCHIVE DESTINATION lib)
install(FILES "${NGHTTP2_SOURCE_DIRECTORY}/lib/includes/nghttp2/nghttp2.h" "${CMAKE_CURRENT_BINARY_DIR}/nghttp2/lib/includes/nghttp2/nghttp2ver.h" DESTINATION include/nghttp2)
