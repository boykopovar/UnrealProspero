cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoOggVorbis LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT EXISTS "${OGG_SOURCE_DIRECTORY}/include/ogg/ogg.h" OR NOT EXISTS "${VORBIS_SOURCE_DIRECTORY}/include/vorbis/codec.h")
    message(FATAL_ERROR "Set OGG_SOURCE_DIRECTORY and VORBIS_SOURCE_DIRECTORY to the Ogg and Vorbis sources supplied with Unreal")
endif()
set(oggConfig "${CMAKE_CURRENT_BINARY_DIR}/include/ogg/config_types.h")
file(GENERATE OUTPUT "${oggConfig}" CONTENT "#ifndef OGG_CONFIG_TYPES_H\n#define OGG_CONFIG_TYPES_H\n#include <stdint.h>\ntypedef int16_t ogg_int16_t;\ntypedef uint16_t ogg_uint16_t;\ntypedef int32_t ogg_int32_t;\ntypedef uint32_t ogg_uint32_t;\ntypedef int64_t ogg_int64_t;\n#endif\n")
add_library(ogg STATIC "${OGG_SOURCE_DIRECTORY}/src/bitwise.c" "${OGG_SOURCE_DIRECTORY}/src/framing.c")
target_include_directories(ogg PUBLIC "${OGG_SOURCE_DIRECTORY}/include" "${CMAKE_CURRENT_BINARY_DIR}/include")
set(vorbisSources mdct smallft block envelope window lsp lpc analysis synthesis psy info floor1 floor0 res0 mapping0 registry codebook sharedbook lookup bitrate)
list(TRANSFORM vorbisSources PREPEND "${VORBIS_SOURCE_DIRECTORY}/lib/")
list(TRANSFORM vorbisSources APPEND ".c")
add_library(vorbis STATIC ${vorbisSources})
add_library(vorbisfile STATIC "${VORBIS_SOURCE_DIRECTORY}/lib/vorbisfile.c")
add_library(vorbisenc STATIC "${VORBIS_SOURCE_DIRECTORY}/lib/vorbisenc.c")
target_include_directories(vorbis PUBLIC "${VORBIS_SOURCE_DIRECTORY}/include")
target_link_libraries(vorbis PUBLIC ogg)
target_link_libraries(vorbisfile PUBLIC vorbis)
target_link_libraries(vorbisenc PUBLIC vorbis)
target_include_directories(vorbisenc PRIVATE "${VORBIS_SOURCE_DIRECTORY}/lib")
foreach(codec ogg vorbis vorbisfile vorbisenc)
    set_target_properties(${codec} PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 99 C_STANDARD_REQUIRED ON)
    target_link_libraries(${codec} PRIVATE AnyLibcxx::Runtime)
    target_compile_options(${codec} PRIVATE -fno-builtin -fno-stack-protector)
endforeach()
install(TARGETS ogg ARCHIVE DESTINATION lib COMPONENT Ogg)
install(TARGETS vorbis vorbisfile vorbisenc ARCHIVE DESTINATION lib COMPONENT Vorbis)
install(FILES "${oggConfig}" "${OGG_SOURCE_DIRECTORY}/include/ogg/ogg.h" "${OGG_SOURCE_DIRECTORY}/include/ogg/os_types.h" DESTINATION include/ogg COMPONENT Ogg)
