cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoOodleArchive LANGUAGES NONE)
foreach(input OODLE_ARCHIVE LIBC_IMPORT_LIBRARY)
    if(NOT DEFINED ${input} OR NOT EXISTS "${${input}}" OR IS_DIRECTORY "${${input}}")
        message(FATAL_ERROR "Set ${input} to an existing archive or import library")
    endif()
endforeach()
find_program(LLVM_NM NAMES llvm-nm REQUIRED)
find_program(LLVM_OBJCOPY NAMES llvm-objcopy REQUIRED)
set(OODLE_COMPONENT core CACHE STRING "Oodle component")
if(NOT OODLE_COMPONENT MATCHES "^(core|network)$")
    message(FATAL_ERROR "OODLE_COMPONENT must be core or network")
endif()
if(OODLE_COMPONENT STREQUAL "network")
    set(library oo2net)
else()
    set(library oo2core)
endif()
option(OODLE_DEBUG "Prepare the debug Oodle archive" OFF)
if(OODLE_DEBUG)
    string(APPEND library "_debug")
endif()
set(output "${CMAKE_CURRENT_BINARY_DIR}/lib${library}.a")
add_custom_command(OUTPUT "${output}"
    COMMAND "${CMAKE_COMMAND}" "-DOODLE_ARCHIVE=${OODLE_ARCHIVE}" "-DOODLE_COMPONENT=${OODLE_COMPONENT}" "-DLIBC_IMPORT_LIBRARY=${LIBC_IMPORT_LIBRARY}" "-DLLVM_NM=${LLVM_NM}" "-DLLVM_OBJCOPY=${LLVM_OBJCOPY}" "-DOUTPUT=${output}" -P "${CMAKE_CURRENT_SOURCE_DIR}/PrepareArchive.cmake"
    DEPENDS "${OODLE_ARCHIVE}" "${LIBC_IMPORT_LIBRARY}" "${LLVM_NM}" "${LLVM_OBJCOPY}" "${CMAKE_CURRENT_SOURCE_DIR}/PrepareArchive.cmake"
    VERBATIM)
add_custom_target(OodleArchive ALL DEPENDS "${output}")
install(FILES "${output}" DESTINATION lib)
