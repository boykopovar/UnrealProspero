cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoOpus LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT EXISTS "${OPUS_SOURCE_DIRECTORY}/BuildForUE/opus/CMakeLists.txt" OR NOT EXISTS "${OPUS_SOURCE_DIRECTORY}/BuildForUE/resampler/CMakeLists.txt")
    message(FATAL_ERROR "Set OPUS_SOURCE_DIRECTORY to the Opus 1.1 sources supplied with Unreal")
endif()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
set(FIXED_POINT OFF)
add_subdirectory("${OPUS_SOURCE_DIRECTORY}/BuildForUE/opus" opus)
add_subdirectory("${OPUS_SOURCE_DIRECTORY}/BuildForUE/resampler" resampler)
foreach(codec opus resampler)
    set_target_properties(${codec} PROPERTIES POSITION_INDEPENDENT_CODE ON C_STANDARD 99 C_STANDARD_REQUIRED ON)
    target_link_libraries(${codec} PRIVATE AnyLibcxx::Runtime)
    target_compile_options(${codec} PRIVATE -fno-builtin -fno-stack-protector)
endforeach()
install(TARGETS opus resampler ARCHIVE DESTINATION lib)
