cmake_minimum_required(VERSION 3.22)
project(UnrealProsperoRuntimeProfile LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
add_library(UnrealProsperoRuntimeProfile OBJECT Probe.cpp)
set_target_properties(UnrealProsperoRuntimeProfile PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_link_libraries(UnrealProsperoRuntimeProfile PRIVATE AnyLibcxx::Runtime)
target_include_directories(UnrealProsperoRuntimeProfile SYSTEM PRIVATE "${ADK_HEADERS}")
separate_arguments(compilerFlags NATIVE_COMMAND "${CMAKE_CXX_FLAGS}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/Compiler.txt" CONTENT "${CMAKE_CXX_COMPILER}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/CompilerId.txt" CONTENT "${CMAKE_CXX_COMPILER_ID}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/Archiver.txt" CONTENT "${CMAKE_AR}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/CompilerTarget.txt" CONTENT "${CMAKE_CXX_COMPILER_TARGET}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/Sysroot.txt" CONTENT "${CMAKE_SYSROOT}\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/CompilerFlags.txt" CONTENT "$<JOIN:${compilerFlags},\n>\n")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/CompileOptions.txt" CONTENT "$<JOIN:$<TARGET_PROPERTY:UnrealProsperoRuntimeProfile,COMPILE_OPTIONS>,\n>\n" TARGET UnrealProsperoRuntimeProfile)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/CompileDefinitions.txt" CONTENT "$<JOIN:$<TARGET_PROPERTY:UnrealProsperoRuntimeProfile,COMPILE_DEFINITIONS>,\n>\n" TARGET UnrealProsperoRuntimeProfile)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/IncludeDirectories.txt" CONTENT "$<JOIN:$<TARGET_PROPERTY:UnrealProsperoRuntimeProfile,INCLUDE_DIRECTORIES>,\n>\n" TARGET UnrealProsperoRuntimeProfile)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/LinkOptions.txt" CONTENT "$<JOIN:$<TARGET_PROPERTY:UnrealProsperoRuntimeProfile,LINK_OPTIONS>,\n>\n" TARGET UnrealProsperoRuntimeProfile)
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/profile/RuntimeLibrary.txt" CONTENT "$<TARGET_FILE:AnyLibcxx::Runtime>\n" TARGET UnrealProsperoRuntimeProfile)
