cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoZlib LANGUAGES C CXX)
find_package(AnyDevKit CONFIG REQUIRED)
if(NOT DEFINED ZLIB_SOURCE_DIRECTORY OR NOT EXISTS "${ZLIB_SOURCE_DIRECTORY}/zlib.h")
    message(FATAL_ERROR "Set ZLIB_SOURCE_DIRECTORY to the zlib sources supplied with Unreal")
endif()
set(sources adler32 compress crc32 deflate gzclose gzlib gzread gzwrite inflate infback inftrees inffast trees uncompr zutil)
list(TRANSFORM sources PREPEND "${ZLIB_SOURCE_DIRECTORY}/")
list(TRANSFORM sources APPEND ".c")
add_library(UnrealProsperoZlib STATIC ${sources} src/File.cpp)
set_target_properties(UnrealProsperoZlib PROPERTIES OUTPUT_NAME z POSITION_INDEPENDENT_CODE ON C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_include_directories(UnrealProsperoZlib PUBLIC "${ZLIB_SOURCE_DIRECTORY}")
target_include_directories(UnrealProsperoZlib PRIVATE include)
target_include_directories(UnrealProsperoZlib SYSTEM PRIVATE "${ADK_HEADERS}")
target_link_libraries(UnrealProsperoZlib PRIVATE AnyLibcxx::Runtime)
target_compile_definitions(UnrealProsperoZlib PRIVATE HAVE_UNISTD_H)
target_compile_definitions(UnrealProsperoZlib PRIVATE "$<$<COMPILE_LANGUAGE:C>:open=UnrealProsperoZlibOpen;close=UnrealProsperoZlibClose;read=UnrealProsperoZlibRead;write=UnrealProsperoZlibWrite>")
if(ADK_TARGET STREQUAL "windows")
    target_compile_definitions(UnrealProsperoZlib PRIVATE "$<$<COMPILE_LANGUAGE:C>:_wopen=UnrealProsperoZlibWideOpen;_lseeki64=UnrealProsperoZlibSeek>")
else()
    target_compile_definitions(UnrealProsperoZlib PRIVATE "$<$<COMPILE_LANGUAGE:C>:lseek=UnrealProsperoZlibSeek;lseek64=UnrealProsperoZlibSeek>")
endif()
target_compile_options(UnrealProsperoZlib PRIVATE -fno-builtin -fno-stack-protector)
install(TARGETS UnrealProsperoZlib ARCHIVE DESTINATION lib)
