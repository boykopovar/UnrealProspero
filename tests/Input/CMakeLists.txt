cmake_minimum_required(VERSION 3.24)
project(AnyUnrealInputTests LANGUAGES CXX)
find_package(AnyDevKit CONFIG REQUIRED)
set(runtime_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/AnyUnreal/Source/Runtime")
adk_add_executable(ControllerCheck Controller.cpp "${runtime_root}/ApplicationCore/Private/AnyUnreal/Input/ControllerHub.cpp")
adk_link_libraries(ControllerCheck libc libkernel libScePad libSceUserService libSceSysmodule)
target_include_directories(ControllerCheck PRIVATE "${runtime_root}/ApplicationCore/Public" "${runtime_root}/Core/Public")
target_compile_definitions(ControllerCheck PRIVATE APPLICATIONCORE_API=)
target_compile_features(ControllerCheck PRIVATE cxx_std_20)
target_compile_options(ControllerCheck PRIVATE -Wall -Wextra -Wpedantic -Werror)
set_target_properties(ControllerCheck PROPERTIES CXX_EXTENSIONS OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
