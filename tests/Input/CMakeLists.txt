cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoInputTests LANGUAGES CXX)
find_package(AnyDevKit CONFIG REQUIRED)
set(runtime_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/UnrealProspero/Source/Runtime")
adk_add_executable(ControllerCheck Controller.cpp "${runtime_root}/ApplicationCore/Private/UnrealProspero/Input/ControllerHub.cpp")
adk_link_libraries(ControllerCheck libc libkernel libScePad libSceUserService libSceSysmodule)
adk_add_executable(ControllerStateCheck State.cpp "${runtime_root}/ApplicationCore/Private/UnrealProspero/Input/ControllerState.cpp")
adk_link_libraries(ControllerStateCheck libc libkernel)
foreach(input_target ControllerCheck ControllerStateCheck)
    target_include_directories(${input_target} PRIVATE "${runtime_root}/ApplicationCore/Public" "${runtime_root}/Core/Public")
    target_compile_definitions(${input_target} PRIVATE APPLICATIONCORE_API=)
    target_compile_features(${input_target} PRIVATE cxx_std_20)
    target_compile_options(${input_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    set_target_properties(${input_target} PROPERTIES CXX_EXTENSIONS OFF RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
endforeach()
