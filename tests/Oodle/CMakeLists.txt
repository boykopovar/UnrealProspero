cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoOodleChecks LANGUAGES CXX)
set(runtime_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/UnrealProspero/Source/Runtime/UnrealProsperoOodleRuntime")
add_executable(CtypeCheck Ctype.cpp "${runtime_root}/Private/UnrealProspero/Oodle/Ctype.cpp")
target_include_directories(CtypeCheck PRIVATE "${runtime_root}/Public")
set_target_properties(CtypeCheck PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_options(CtypeCheck PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME oodle_ctype COMMAND CtypeCheck C C.UTF-8)
