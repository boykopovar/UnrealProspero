cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoOodleChecks LANGUAGES CXX)
option(UNREALPROSPERO_TEST_OODLE_RUNTIME "Run Oodle through the AnyDevKit runtime" OFF)
if(UNREALPROSPERO_TEST_OODLE_RUNTIME)
    foreach(input OODLE_INCLUDE_DIRECTORY OODLE_ARCHIVE)
        if(NOT DEFINED ${input} OR NOT EXISTS "${${input}}")
            message(FATAL_ERROR "Set ${input} to the Oodle SDK headers or prepared archive")
        endif()
    endforeach()
    find_package(AnyDevKit CONFIG REQUIRED)
    adk_add_executable(OodleCheck RoundTrip.cpp)
    set_target_properties(OodleCheck PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
    target_include_directories(OodleCheck SYSTEM PRIVATE "${OODLE_INCLUDE_DIRECTORY}")
    target_link_libraries(OodleCheck PRIVATE "${OODLE_ARCHIVE}")
    adk_link_libraries(OodleCheck libc libkernel)
    return()
endif()
set(runtime_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/UnrealProspero/Source/Runtime/UnrealProsperoOodleRuntime")
add_executable(CtypeCheck Ctype.cpp "${runtime_root}/Private/UnrealProspero/Oodle/Ctype.cpp")
target_include_directories(CtypeCheck PRIVATE "${runtime_root}/Public")
set_target_properties(CtypeCheck PROPERTIES CXX_STANDARD 20 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_options(CtypeCheck PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME oodle_ctype COMMAND CtypeCheck C C.UTF-8)
