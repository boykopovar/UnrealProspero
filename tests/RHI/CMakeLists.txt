cmake_minimum_required(VERSION 3.24)
project(AnyUnrealShaderLoadingTests LANGUAGES CXX)

set(runtime_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/AnyUnreal/Source/Runtime")
option(ANYUNREAL_TEST_CREATE_SHADER "Create AGC shaders through the release runtime" OFF)
if(ANYUNREAL_TEST_CREATE_SHADER)
    find_package(AnyDevKit CONFIG REQUIRED)
    adk_add_executable(AgcHeaderCheck AgcHeader.cpp
        "${runtime_root}/AnyUnrealRHI/Private/AnyUnreal/RHI/Shader/AgcShader.cpp"
        "${runtime_root}/Core/Private/AnyUnreal/Kernel/DirectAllocation.cpp")
    adk_link_libraries(AgcHeaderCheck libc libkernel libSceAgc)
    set_target_properties(AgcHeaderCheck PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/bin")
    target_compile_definitions(AgcHeaderCheck PRIVATE ANYUNREAL_TEST_CREATE_SHADER=1)
    if(ADK_TARGET STREQUAL "windows")
        target_compile_definitions(AgcHeaderCheck PRIVATE ANYUNREAL_TEST_AGC_TARGET=Windows)
    elseif(ADK_TARGET STREQUAL "linux")
        target_compile_definitions(AgcHeaderCheck PRIVATE ANYUNREAL_TEST_AGC_TARGET=Linux)
    elseif(ADK_TARGET STREQUAL "ps5")
        target_compile_definitions(AgcHeaderCheck PRIVATE ANYUNREAL_TEST_AGC_TARGET=Ps5)
    else()
        message(FATAL_ERROR "Unknown AGC test platform")
    endif()
else()
    if(NOT EXISTS "${ADK_HEADERS}/SceShaders.hpp")
        message(FATAL_ERROR "ADK_HEADERS must point to AnyPS5 public release headers")
    endif()
    add_executable(AgcHeaderCheck AgcHeader.cpp)
    target_include_directories(AgcHeaderCheck SYSTEM PRIVATE "${ADK_HEADERS}")
    target_compile_definitions(AgcHeaderCheck PRIVATE ANYUNREAL_TEST_CREATE_SHADER=0)
endif()
target_sources(AgcHeaderCheck PRIVATE "${runtime_root}/AnyUnrealRHI/Private/AnyUnreal/RHI/Shader/AgcHeader.cpp")
target_include_directories(AgcHeaderCheck PRIVATE "${runtime_root}/AnyUnrealRHI/Public" "${runtime_root}/AnyUnrealShaderCommon/Public" "${runtime_root}/Core/Public")
target_compile_definitions(AgcHeaderCheck PRIVATE ANYUNREALRHI_API= ANYUNREALSHADERCOMMON_API=)
target_compile_features(AgcHeaderCheck PRIVATE cxx_std_20)
set_target_properties(AgcHeaderCheck PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(AgcHeaderCheck PRIVATE /W4 /WX)
else()
    target_compile_options(AgcHeaderCheck PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
