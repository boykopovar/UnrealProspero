cmake_minimum_required(VERSION 3.24)
project(UnrealProsperoShaderReflectionTests LANGUAGES C CXX)

if(WIN32)
    set(CMAKE_OBJECT_PATH_MAX 180)
endif()

if(NOT IS_DIRECTORY "${UNREAL_ROOT}/Engine/Source/ThirdParty/SPIRV-Reflect/SPIRV-Reflect")
    message(FATAL_ERROR "UNREAL_ROOT must provide UE SPIRV-Reflect sources")
endif()

set(common_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/UnrealProspero/Source/Runtime/UnrealProsperoShaderCommon")
enable_testing()
add_executable(ShaderBinary Binary.cpp "${common_root}/Private/Shader/ConstantData.cpp" "${common_root}/Private/Shader/ShaderBinary.cpp")
target_include_directories(ShaderBinary PRIVATE "${common_root}/Public")
target_compile_definitions(ShaderBinary PRIVATE UNREALPROSPEROSHADERCOMMON_API=)
target_compile_features(ShaderBinary PRIVATE cxx_std_20)
set_target_properties(ShaderBinary PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(ShaderBinary PRIVATE /W4 /WX)
else()
    target_compile_options(ShaderBinary PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME shader_binary COMMAND ShaderBinary)

set(reflect_root "${UNREAL_ROOT}/Engine/Source/ThirdParty/SPIRV-Reflect/SPIRV-Reflect")
set(compiler_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/UnrealProspero/Source/Developer/UnrealProsperoShaderCompiler")
add_library(UnrealSpirvReflect STATIC "${reflect_root}/spirv_reflect.c")
target_include_directories(UnrealSpirvReflect SYSTEM PUBLIC "${reflect_root}")
add_executable(Reflect Reflect.cpp "${compiler_root}/Private/Shader/DescriptorLayout.cpp" "${compiler_root}/Private/Shader/SpirvReflection.cpp" "${compiler_root}/Private/Shader/VertexInputs.cpp")
target_include_directories(Reflect PRIVATE "${compiler_root}/Public" "${common_root}/Public")
target_compile_definitions(Reflect PRIVATE UNREALPROSPEROSHADERCOMPILER_API= UNREALPROSPEROSHADERCOMMON_API=)
target_compile_features(Reflect PRIVATE cxx_std_20)
set_target_properties(Reflect PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(Reflect PRIVATE UnrealSpirvReflect)
add_executable(PipelineDescriptors PipelineDescriptors.cpp "${compiler_root}/Private/Shader/PipelineDescriptors.cpp" "${compiler_root}/Private/Shader/DescriptorLayout.cpp" "${compiler_root}/Private/Shader/SpirvReflection.cpp")
target_include_directories(PipelineDescriptors PRIVATE "${compiler_root}/Public" "${common_root}/Public")
target_compile_definitions(PipelineDescriptors PRIVATE UNREALPROSPEROSHADERCOMPILER_API= UNREALPROSPEROSHADERCOMMON_API=)
target_compile_features(PipelineDescriptors PRIVATE cxx_std_20)
set_target_properties(PipelineDescriptors PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(PipelineDescriptors PRIVATE UnrealSpirvReflect)
target_sources(PipelineDescriptors PRIVATE "${compiler_root}/Private/Shader/EntryPoint.cpp")
if(MSVC)
    target_compile_options(PipelineDescriptors PRIVATE /W4 /WX)
else()
    target_compile_options(PipelineDescriptors PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(MSVC)
    target_compile_options(Reflect PRIVATE /W4 /WX)
else()
    target_compile_options(Reflect PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
