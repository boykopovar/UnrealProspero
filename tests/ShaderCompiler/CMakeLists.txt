cmake_minimum_required(VERSION 3.24)
project(AnyUnrealShaderReflectionTests LANGUAGES C CXX)

if(NOT IS_DIRECTORY "${UNREAL_ROOT}/Engine/Source/ThirdParty/SPIRV-Reflect/SPIRV-Reflect")
    message(FATAL_ERROR "UNREAL_ROOT must provide UE SPIRV-Reflect sources")
endif()

set(common_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/AnyUnreal/Source/Runtime/AnyUnrealShaderCommon")
enable_testing()
add_executable(ShaderBinary Binary.cpp "${common_root}/Private/AnyUnreal/Shader/ConstantData.cpp" "${common_root}/Private/AnyUnreal/Shader/ShaderBinary.cpp")
target_include_directories(ShaderBinary PRIVATE "${common_root}/Public")
target_compile_definitions(ShaderBinary PRIVATE ANYUNREALSHADERCOMMON_API=)
target_compile_features(ShaderBinary PRIVATE cxx_std_20)
set_target_properties(ShaderBinary PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
    target_compile_options(ShaderBinary PRIVATE /W4 /WX)
else()
    target_compile_options(ShaderBinary PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME shader_binary COMMAND ShaderBinary)

set(reflect_root "${UNREAL_ROOT}/Engine/Source/ThirdParty/SPIRV-Reflect/SPIRV-Reflect")
set(compiler_root "${CMAKE_CURRENT_SOURCE_DIR}/../../Engine/Platforms/AnyUnreal/Source/Developer/AnyUnrealShaderCompiler")
add_library(UnrealSpirvReflect STATIC "${reflect_root}/spirv_reflect.c")
target_include_directories(UnrealSpirvReflect SYSTEM PUBLIC "${reflect_root}")
add_executable(Reflect Reflect.cpp "${compiler_root}/Private/AnyUnreal/Shader/DescriptorLayout.cpp")
target_include_directories(Reflect PRIVATE "${compiler_root}/Public" "${common_root}/Public")
target_compile_definitions(Reflect PRIVATE ANYUNREALSHADERCOMPILER_API= ANYUNREALSHADERCOMMON_API=)
target_compile_features(Reflect PRIVATE cxx_std_20)
set_target_properties(Reflect PROPERTIES CXX_EXTENSIONS OFF)
target_link_libraries(Reflect PRIVATE UnrealSpirvReflect)
if(MSVC)
    target_compile_options(Reflect PRIVATE /W4 /WX)
else()
    target_compile_options(Reflect PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
